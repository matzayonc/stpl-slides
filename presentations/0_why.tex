\section{Problem z prywatnymi serwerami}

\begin{frame}{Prywatne serwery}
  \begin{itemize}
    \item Twoje dane, pieniądze, historia --- na serwerze firmy
    \item Firma może \textbf{zmienić zasady} lub zablokować konto
    \item Kod produkcyjny może różnić się od audytowanego
  \end{itemize}
  \vspace{0.4cm}
  \begin{exampleblock}{Przykłady z życia}
    \begin{itemize}
      \item PayPal zamraża konta twórców bez ostrzeżenia\\
      \item Giełda FTX wypłacała środki klientów --- do czasu\\
      \item AWS outage wyłącza dziesiątki popularnych serwisów naraz
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{Problem zaufania}
  \begin{columns}
    \column{0.48\textwidth}
      \begin{block}{Co widzisz}
        \begin{itemize}
          \item Aplikacja działa
          \item Saldo się zgadza
          \item Historia transakcji jest dostępna
        \end{itemize}
      \end{block}

    \column{0.48\textwidth}
      \begin{alertblock}{Czego nie wiesz}
        \begin{itemize}
          \item Czy dane nie zostały zmienione
          \item Czy środki faktycznie tam są
          \item Co się stanie jutro
        \end{itemize}
      \end{alertblock}
  \end{columns}
\end{frame}

\section{Sieci z konsensusem BFT}

\begin{frame}{Wspólna sieć}
  \begin{itemize}
    \item Tysiące niezależnych węzłów przechowuje ten sam stan
    \item Żeby zmienić dane, musisz przekonać \textbf{większość sieci}
    \item Nieszczery węzeł zostaje przegłosowany
    \item To jest \textbf{Byzantine Fault Tolerance (BFT)}
  \end{itemize}
  \vspace{0.3cm}
  \begin{exampleblock}{Efekt}
    Kod uruchomiony w takiej sieci wykona się \textbf{dokładnie tak jak napisany} ---
    bez możliwości ingerencji żadnego pojedynczego podmiotu.
  \end{exampleblock}
\end{frame}

\begin{frame}{Rejestr BFT}
  \begin{itemize}
    \item Zmiana bloku kryptograficznie unieważnia kolejne
    \item Stan sieci jest \textbf{publiczny} --- każdy może go zweryfikować
    \item Każda operacja \textbf{podpisana kluczem} właściciela
  \end{itemize}
  \vspace{0.3cm}
  \begin{columns}
    \column{0.48\textwidth}
      \begin{block}{Prywatny serwer}
        Zaufanie do właściciela
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Blockchain}
        Zaufanie do matematyki
      \end{block}
  \end{columns}
\end{frame}

\section{Zastosowanie w finansach}

\begin{frame}{Smart kontrakty}
  \begin{itemize}
    \item Logika finansowa to \textbf{kod} na blockchainie
    \item \textbf{Smart kontrakt} zastępuje bank, giełdę, notariusza
    \item Warunki jawne i egzekwowane automatycznie
    \item Permissionless --- bez dokumentów i zgody operatora
  \end{itemize}
  \vspace{0.3cm}
  \begin{exampleblock}{Przykłady}
    \textbf{Pożyczka}: kod sprawdza zabezpieczenie i wypłaca środki bez banku\\
    \textbf{Wymiana walut}: algorytm wycenia i realizuje transakcję bez brokera
  \end{exampleblock}
\end{frame}

\section{Dlaczego Solana?}

\begin{frame}{Obecne ograniczenia}
  \begin{itemize}
    \item \textbf{Bitcoin} --- BFT i bezpieczny, ale bez smart kontraktów
    \item \textbf{Ethereum} --- 15 TPS i gas \textgreater{} \$50 przy obciążeniu
    \item Wysoki koszt = tylko duże operacje mają sens
    \item Layer 2 obniżają koszty, ale fragmentują ekosystem
  \end{itemize}
  \vspace{0.3cm}
  \begin{alertblock}{Problem}
    BFT compute bez skali to zabawka, nie infrastruktura.
  \end{alertblock}
\end{frame}

\begin{frame}{Solana}
  \begin{block}{Kluczowe liczby}
    \begin{itemize}
      \item \textbf{50\,000+ TPS} --- przepustowość porównywalna z Visa
      \item \textbf{\textless{} \$0.001} --- niezależnie od obciążenia sieci
      \item \textbf{400 ms} --- czas finalizacji bloku
      \item \textbf{Jeden layer} --- bez fragmentacji ekosystemu
    \end{itemize}
  \end{block}
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{PoH} --- kryptograficzny zegar dla węzłów
    \item \textbf{Sealevel} --- równoległe wykonanie transakcji
  \end{itemize}
\end{frame}
