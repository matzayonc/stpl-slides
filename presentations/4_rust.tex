\section{Po co Rust?}

\begin{frame}{Nie uczymy Rusta}
  \begin{itemize}
    \item Pełny Rust to semestralny kurs --- nie zmieścimy go w lekcji
    \item Cel jest węższy --- odblokować czytanie \texttt{lib.rs} Anchora
    \item Lifetime'y, generyki, \texttt{async} i borrow checker w głębi --- pomijamy
    \item Po lekcji rozpoznajesz \texttt{mut}, \texttt{Option}, \texttt{Result}, makra
  \end{itemize}
  \vspace{0.25cm}
  \begin{alertblock}{To nie kurs RUST}
    Pełny kurs Rusta odbędzie się oddzielnie --- tutaj dajemy minimum
    niezbędne do Anchora.
  \end{alertblock}
\end{frame}

\begin{frame}{Kod on-chain}
  \begin{itemize}
    \item Bug w programie Solany to \textbf{strata użytkowników} --- transakcji nie cofniesz
    \item Większość exploitów to klasy błędów, które typowany język wyklucza w kompilacji
    \item Rust gwarantuje brak \texttt{null}, brak wyścigów danych i jawność błędów --- przed uruchomieniem
  \end{itemize}
  \vspace{0.3cm}
  \begin{alertblock}{Skutek}
    Cała klasa bugów staje się \textbf{niemożliwa} zanim kod dotrze na sieć.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Jak wygląda}

fn deposit(amount: u64, owner: Pubkey) -> Result<()> {

\section{Wzorce Rusta}

\begin{frame}[fragile]{Niemutowalność}
  \begin{itemize}
    \item W JS każda zmienna jest mutowalna --- w Ruście \textbf{odwrotnie}
    \item Jak arkusz z ochroną komórek: jawna akcja jest wymagana do edycji
    \item \texttt{mut} to świadoma decyzja widoczna w kodzie; kompilator jej pilnuje
  \end{itemize}
  \begin{block}{let vs let mut}
\begin{lstlisting}
let x = 5;      // niemutowalne (domyslnie)
// x = 6;       // BLAD kompilacji

let mut y = 5;  // jawna zgoda na zmiane
y = 6;          // OK
\end{lstlisting}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Brak klas}
  \begin{itemize}
    \item Rust nie ma \texttt{class} --- są \texttt{struct} (pola) i \texttt{impl} (metody)
    \item Bez dziedziczenia --- zamiast tego \textbf{traity}: interfejsy on steroids
    \item Każdy typ z Anchora to \texttt{struct} z blokiem \texttt{impl}
  \end{itemize}
  \begin{block}{struct + impl}
\begin{lstlisting}
struct Licznik {
    wartosc: u64,
}

impl Licznik {
    fn inkrementuj(&mut self) {
        self.wartosc += 1;
    }
}
\end{lstlisting}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Typy w Anchorze}
  \begin{itemize}
    \item \texttt{u8}, \texttt{u64}, \texttt{i64} --- liczby; \texttt{u} = unsigned (bez znaku)
    \item \texttt{Pubkey} --- 32-bajtowy adres konta lub właściciela
    \item \texttt{String}, \texttt{bool}, \texttt{Vec<T>} --- tekst, prawda/fałsz, lista
  \end{itemize}
  \begin{block}{Typowy stan konta}
\begin{lstlisting}
#[account]
pub struct Stan {
    pub wlasciciel: Pubkey,       // adres
    pub saldo:      u64,          // liczba dodatnia
    pub aktywny:    bool,         // flaga
    pub glosy:      Vec<Pubkey>,  // lista adresow
}
\end{lstlisting}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Match}
  \begin{itemize}
    \item \texttt{switch} w JS milcząco pomija przypadki --- błąd widoczny dopiero w runtime
    \item Jak lista kontrolna pilota: procedura jest niekompletna, dopóki każdy punkt nie jest omówiony
    \item \texttt{match} wymaga pokrycia \textbf{wszystkich} gałęzi --- brak = błąd kompilacji
  \end{itemize}
  \begin{alertblock}{Match wyczerpujący}
\begin{lstlisting}
match status {
    0 => msg!("nowy"),
    1 => msg!("aktywny"),
    _ => msg!("inny"),   // _ = pozostale przypadki
}                        // brak _ przy >1 wariancie = BLAD
\end{lstlisting}
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Option --- bez null}
  \begin{itemize}
    \item \texttt{null} rzuca wyjątek dopiero przy dostępie --- za późno
    \item Jak paczka kurierska: albo \texttt{Some(paczka)} na progu, albo \texttt{None} z awizo --- nie ma ciszy
    \item Kompilator wymusza obsługę \texttt{None} \textbf{przed} dostępem do wartości
  \end{itemize}
  \begin{exampleblock}{Option\textless{}T\textgreater{}}
\begin{lstlisting}
let first = accounts.first();   // Option<&Account>
match first {
    Some(acc) => msg!("konto: {}", acc.key()),
    None      => return Err(MyError::NoAccount.into()),
}
\end{lstlisting}
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Result --- bez wyjątków}
  \begin{itemize}
    \item \texttt{try/catch} ukrywa, które funkcje mogą zawieść --- trzeba czytać dokumentację
    \item W Ruście błąd jest \textbf{wartością} widoczną w sygnaturze --- nie da się go przeoczyć
    \item Operator \texttt{?} propaguje błąd w górę bez zagnieżdżonych \texttt{if}-ów
  \end{itemize}
  \begin{alertblock}{Result\textless{}T, E\textgreater{}}
\begin{lstlisting}
fn load(ctx: &Context<Withdraw>) -> Result<u64> {
    let amt = ctx.accounts.vault.balance()?;
    // ? = jesli Err, zwroc blad wywolujacemu
    Ok(amt)
}
\end{lstlisting}
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Makra}
  \begin{itemize}
    \item Makro to \textbf{kod, który pisze kod} --- rozwijane przed kompilacją
    \item Jak formularz z polami do uzupełnienia: makro wstawia boilerplate, autorowi zostaje logika
    \item \texttt{\#[...]} to atrybut; \texttt{!} po nazwie oznacza wywołanie makra
  \end{itemize}
  \begin{exampleblock}{Trzy formy}
\begin{lstlisting}
#[program]                        // atrybut: ozdabia modul
pub mod vault { /* ... */ }

msg!("wplacono: {}", amount);     // makro logujace
let ids = vec![pk_a, pk_b];      // makro tworzace Vec
\end{lstlisting}
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Pożyczanie}
  \begin{itemize}
    \item Rust nie kopiuje danych domyślnie --- przekazuje \textbf{pożyczkę} referencji (\texttt{\&})
    \item Jak biblioteka: \texttt{\&} = czytam książkę, \texttt{\&mut} = piszę notatki; właściciel się nie zmienia
    \item Kompilator gwarantuje: jeden \texttt{\&mut} \textbf{lub} wiele \texttt{\&} --- nigdy oba naraz
    \item \textcolor{stplmuted}{Tu --- intuicja, nie teoria; pełny borrow checker na kursie RUST}
  \end{itemize}
  \begin{block}{Dwa rodzaje pożyczki}
\begin{lstlisting}
fn read_vault(v: &Vault)            { msg!("{}", v.balance); }
fn add_funds(v: &mut Vault, n: u64) { v.balance += n; }

let vault = &mut ctx.accounts.vault;  // typowy wzorzec Anchor
\end{lstlisting}
  \end{block}
\end{frame}

\section{W praktyce}

\begin{frame}[fragile]{Czytamy razem}
  \begin{itemize}
    \item \texttt{struct Deposit}, \texttt{Context<T>}, \texttt{\&mut} pożyczka pola (\texttt{mut})
    \item \texttt{checked\_add} zwraca \texttt{Option}; \texttt{ok\_or(...)?} zamienia na \texttt{Result}
    \item \texttt{msg!} makro, \texttt{Ok(())} jawny sukces
    \item \textcolor{stplmuted}{Snippet z Anchora --- szczegóły frameworku w lekcji 6}
  \end{itemize}
  \begin{block}{deposit --- vault program}
\begin{lstlisting}
pub fn deposit(ctx: Context<Deposit>,    // Context<T>
               amount: u64) -> Result<()> {
    let vault = &mut ctx.accounts.vault; // pozyczka &mut
    vault.balance = vault.balance
        .checked_add(amount)             // zwraca Option
        .ok_or(VaultError::Overflow)?;   // Option->Result, ?
    msg!("Wplacono: {}", amount);        // makro
    Ok(())                               // Result: sukces
}
\end{lstlisting}
  \end{block}
\end{frame}

% ================================================================
\begin{frame}{Podsumowanie}
  \begin{enumerate}
    \item \textbf{Niemutowalność} domyślna --- \texttt{mut} = jawna decyzja
    \item \textbf{struct + impl} zamiast klas; traity zamiast dziedziczenia
    \item \textbf{Option} zamiast null; \textbf{Result} zamiast wyjątków, \texttt{?} propaguje
    \item \textbf{Pożyczanie} (\texttt{\&}, \texttt{\&mut}) --- bez kopiowania, bez wyścigów
    \item \textbf{Makra} (\texttt{\#[...]}, \texttt{msg!}) generują boilerplate
  \end{enumerate}
  \vspace{0.2cm}
  \begin{exampleblock}{Efekt}
    Nie znasz jeszcze całego Rusta --- ale \texttt{lib.rs} z Anchora czytasz bez paniki.
  \end{exampleblock}
\end{frame}
