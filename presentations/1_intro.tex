\section{Czym jest Solana?}

\begin{frame}{Czym jest Solana?}
  \begin{itemize}
    \item Rozproszona sieć ze środowiskiem wykonawczym - chmura
    \item Blockchain wysokiej wydajności --- 50\,000+ TPS - szybka
    \item Proof of History (PoH) + Proof of Stake (PoS) - bezpieczna
    \item Niskie opłaty transakcyjne (\textless{} \$0.001) - tania
  \end{itemize}
\end{frame}

% ----------------------------------------------------------------
\section{Czym jest Rust?}

\begin{frame}{Czym jest Rust?}
  \begin{itemize}
    \item Główny język programów on-chain na Solanie
    \item Język systemowy: bezpieczeństwo pamięci bez GC
    \item Rozbudowany ekosystem: biblioteki, narzędzia, materiały
    \item Narzędzia: \texttt{rustup}, \texttt{cargo}, \texttt{clippy}
  \end{itemize}
\end{frame}

\section{Co z danymi?}

\begin{frame}{Co z danymi?}
  \begin{itemize}
    \item Wszystko jest \textbf{kontem} (\emph{account})
    \item Dane przechowywane w polach konta (do 10 MB)
    \item Rent --- opłata za zajmowanie przestrzeni
    \item PDAs --- deterministyczne adresy dla programów
  \end{itemize}
\end{frame}

% ----------------------------------------------------------------
\section{A logowanie?}

\begin{frame}{A logowanie?}
  \begin{itemize}
    \item Klucze ed25519 --- para: klucz prywatny + publiczny
    \item Klucz publiczny = adres konta w sieci
    \item Każda transakcja podpisana kluczem prywatnym
    \item Portfel (wallet) przechowuje klucz i podpisuje w imieniu użytkownika
  \end{itemize}
\end{frame}

\section{Czym jest dApp?}
